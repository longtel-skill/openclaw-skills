## 命令运行示例

（在技能根目录下执行）

    python3 scripts/generate_doc.py \
      --company "示例科技有限公司" \
      --credit-code "91110000MA0000000X" \
      --address "北京市示例区示例路 1 号" \
      --business "信息技术服务、计算机系统服务" \
      --year "2023" \
      --invoice-count 1 \
      --invoice-amount 10000.00 \
      --invoice-tax 1300.00 \
      --conclusion "经核查，该发票为我司正常经营业务取得，交易真实有效。" \
      --tax-office "示例区税务局" \
      --output "test-case1"
